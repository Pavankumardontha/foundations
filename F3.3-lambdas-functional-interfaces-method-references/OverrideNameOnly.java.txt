// Is the method NAME alone enough for @Override to be satisfied?

class Printer {
    void print(String text) {
        System.out.println("parent prints: " + text);
    }
}

class FancyPrinter extends Printer {
    @Override
    void print(int copies) {     // SAME NAME 'print', but different parameter type
        System.out.println("printing " + copies + " copies");
    }
}

public class OverrideNameOnly {
    public static void main(String[] args) {
        new FancyPrinter().print(3);
    }
}
